cmake_minimum_required(VERSION 3.16)
project(cardgame_rx VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
add_library(cards src/model.c src/layout.c src/scene.c)
target_include_directories(cards PUBLIC src)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
add_executable(card-preview src/preview.c src/render.c)
target_link_libraries(card-preview PRIVATE cards PkgConfig::SDL2)
enable_testing()
add_executable(foundation-tests tests/foundation.c)
target_link_libraries(foundation-tests PRIVATE cards)
add_test(NAME foundation COMMAND foundation-tests)
add_executable(rendering-tests tests/rendering.c src/render.c)
target_link_libraries(rendering-tests PRIVATE cards PkgConfig::SDL2)
add_test(NAME rendering COMMAND rendering-tests)
