cmake_minimum_required(VERSION 3.16)
project(cardgame_rx VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
add_library(cards src/model.c src/layout.c src/scene.c)
target_include_directories(cards PUBLIC src)
enable_testing()
add_executable(foundation-tests tests/foundation.c)
target_link_libraries(foundation-tests PRIVATE cards)
add_test(NAME foundation COMMAND foundation-tests)

option(CARDGAME_BUILD_PREVIEW "Build SDL desktop preview and rendering tests" ON)
if(CARDGAME_BUILD_PREVIEW)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    add_executable(card-preview src/preview.c src/render.c)
    target_link_libraries(card-preview PRIVATE cards PkgConfig::SDL2)
    add_executable(rendering-tests tests/rendering.c src/render.c)
    target_link_libraries(rendering-tests PRIVATE cards PkgConfig::SDL2)
    add_test(NAME rendering COMMAND rendering-tests)
endif()

# Pure rules have no SDL linkage; the existing static preview stays unchanged.
add_library(klondike src/klondike.c)
target_link_libraries(klondike PUBLIC cards)
add_executable(klondike-tests tests/klondike.c)
target_link_libraries(klondike-tests PRIVATE klondike)
add_test(NAME klondike COMMAND klondike-tests)

add_library(play src/play.c)
target_link_libraries(play PUBLIC klondike)
add_executable(play-tests tests/play.c)
target_link_libraries(play-tests PRIVATE play)
add_test(NAME play COMMAND play-tests)
add_library(play-view src/play_view.c)
target_link_libraries(play-view PUBLIC play)
if(CARDGAME_BUILD_PREVIEW)
    add_library(play-desktop src/play_render.c src/keyboard.c src/desktop.c src/render.c)
    target_link_libraries(play-desktop PUBLIC play-view PkgConfig::SDL2)
    add_executable(cardgame src/game.c)
    target_link_libraries(cardgame PRIVATE play-desktop)
endif()
if(CARDGAME_BUILD_PREVIEW)
    add_executable(play-desktop-tests tests/play_desktop.c)
    target_link_libraries(play-desktop-tests PRIVATE play-desktop)
    add_test(NAME play-desktop COMMAND play-desktop-tests)
endif()
